# Deploy-root requirements for Stage 1 on Azure App Service.
# Oryx installs THIS file (at the deployed root = the governance_agent folder).
# Stage 1 runs BOTH processes from one App Service (see startup.sh):
#   - gateway/       the public Governance Gateway (PEP) -> imports governance_core/
#   - mcp-minierp/   the consolidated backend         -> imports minierp_core/
# so this single requirements file must cover both.

# MCP -- gateway is a server (to callers) AND client (to the backend); backend is a server.
# Pinned exact, not >=: an unbounded floating version let Oryx install whatever
# mcp release is newest at deploy time, which broke FastMCP's import path and
# renamed mcp.client.streamable_http.streamablehttp_client out from under this
# code (verified working at this exact version, 2026-07-28).
mcp==1.28.1

# Exact-pinned below, not >=: an unbounded mcp>=1.10.0 is what let Oryx install
# a newer, incompatible release on 2026-07-28 and crash-loop the whole site (see
# above). Every direct dependency here gets the same treatment so the next
# floating-version break doesn't repeat it. Versions are what's verified running
# (locally and in prod) as of 2026-07-28; bump deliberately, not by omission.

# ASGI runtime (uvicorn runs both app:app processes)
uvicorn[standard]==0.51.0
starlette==1.3.1

# httpx -- MCP Streamable HTTP client (gateway->backend) + the miniERP GraphQL client
httpx==0.28.1

# Env loading (no-op in prod; App Service app settings supply real env vars)
python-dotenv==1.2.2

# LLM client for the dashboard chat orchestrator (self-hosted Qwen via sglang,
# OpenAI-compatible). Imported lazily; chat degrades gracefully if unset.
openai==2.46.0

# Backup chat LLM (Claude Sonnet via an Anthropic-compatible endpoint), used
# when USE_LOCAL_LLM=false. Imported lazily -- only import-checked, not exercised
# against a live endpoint, since that needs real credentials.
anthropic==0.120.2

# Cosmos DB (Core/SQL API) policy store -- durable, multi-instance per-user policy
# + category scope. Imported lazily by get_store() only when GOVERNANCE_COSMOS_* is
# set; file/local store need no extra deps.
azure-cosmos==4.16.2

# Azure AI Search -- mcp-knowledge's direct read-only query path (search_knowledge/
# answer_from_knowledge), when AZURE_SEARCH_* is set. Imported lazily; the HTTP-proxy
# and local-fallback tiers need no extra deps.
azure-search-documents==12.0.0
azure-core==1.41.0
